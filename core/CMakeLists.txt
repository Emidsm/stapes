# Each module is its own target, and a module can only include what it links.
# That makes the dependency direction a build error to violate:
#   base -> math -> kinematics -> dynamics
#   hardware <- plant (the simulated robot)
#   hardware <- control (cannot see plant, so it cannot tell it is simulated)
#   hardware <- safety (cannot see control or plant, so it shares no state with them)
#   hardware <- faults (sits between software and hardware, implementing the same interfaces)
#   kinematics <- task (plans the motion; knows nothing of hardware)
#   sim -> everything (the one place they are assembled)
add_subdirectory(base)
add_subdirectory(math)
add_subdirectory(kinematics)
add_subdirectory(dynamics)
add_subdirectory(hardware)
add_subdirectory(plant)
add_subdirectory(control)
add_subdirectory(task)
add_subdirectory(safety)
add_subdirectory(faults)
add_subdirectory(sim)
